cmake_minimum_required(VERSION 3.20)
project(steins-terminal VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets DBus)
find_package(qtermwidget6 REQUIRED)
add_executable(steins-terminal-v3 src/main.cpp)
target_link_libraries(steins-terminal-v3 PRIVATE Qt6::Widgets Qt6::DBus qtermwidget6)
target_compile_options(steins-terminal-v3 PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS steins-terminal-v3 RUNTIME DESTINATION bin)
include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    add_executable(steins-terminal-tests src/main.cpp)
    target_compile_definitions(steins-terminal-tests PRIVATE STEINS_TERMINAL_TEST)
    target_link_libraries(steins-terminal-tests PRIVATE Qt6::Widgets Qt6::DBus Qt6::Test qtermwidget6)
    add_test(NAME terminal-engine-contracts COMMAND steins-terminal-tests)
    set_tests_properties(terminal-engine-contracts PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
